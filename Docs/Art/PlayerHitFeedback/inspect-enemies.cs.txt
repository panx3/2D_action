using System;
using System.Linq;
using UnityEditor;
using UnityEngine;
public static class InspectPlayerHitEnemies
{
    public static string Run()
    {
        string[] names = { "TekkyuEnemy", "SpringFrog", "Hachidrill", "WrenchCrab", "BeamScorpion" };
        return string.Join(" | ", names.Select(name => {
            var prefab = AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Enemies/" + name + ".prefab");
            return name + ":" + (prefab == null ? "MISSING" : string.Join(",", prefab.GetComponents<Component>().Select(c => c.GetType().Name)));
        }));
    }
}
